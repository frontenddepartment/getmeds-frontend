User-agent: *
Allow: /

# Admin back office / dashboards. Every admin panel is proxied through this
# domain (vercel.json rewrites to admin.getmeds.ph), all under paths starting
# with /admin or /master. Matched by those prefixes on purpose: robots.txt is
# public, so listing each panel's exact path here would publish them. No
# public page starts with either prefix.
Disallow: /admin
Disallow: /master

# Signed-in and per-visitor pages: account, cart and site search results.
# Nothing here is useful in search, and search result URLs are unbounded.
# (Each also carries <meta name="robots" content="noindex">.)
Disallow: /profile
Disallow: /edit-profile
Disallow: /cart
Disallow: /search

# Business cards. Each of these pages carries a named staff member's mobile
# number, published for whoever is holding the printed card and for nobody
# running a search. Said three times over — here, in <meta name="robots"> in
# business-card.html, and as an X-Robots-Tag header in vercel.json — because
# a Disallow only stops the crawl, not the listing: a page linked from
# elsewhere can still surface in results on the strength of that link alone,
# and the meta and header are what actually keep it out of the index.
Disallow: /card/

# URL sitemap (index of the page, product and blog sitemaps) and the separate
# image sitemap. Both generated by scripts/generate-sitemap.cjs. The
# human-readable sitemap is at https://getmeds.ph/sitemap
Sitemap: https://getmeds.ph/sitemap.xml
Sitemap: https://getmeds.ph/image-sitemap.xml
